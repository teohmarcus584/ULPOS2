async function getNextIRSInvoiceNumber() { try { console.log('📋 Getting next IRS invoice number...'); // ✅ Get ALL invoice numbers - NO LIMIT const { data, error } = await supabase .from('irs_reporting') .select('invoice_no') .not('invoice_no', 'is', null); if (error) { console.error('❌ Error getting invoice numbers:', error); const timestamp = Date.now(); return `INV-${timestamp}`; } let highestNum = 0; if (data && data.length > 0) { console.log(`📋 Found ${data.length} existing invoice numbers`); // ✅ Loop through EVERY invoice to find the highest number for (let i = 0; i < data.length; i++) { const invoice = data[i].invoice_no; if (!invoice) continue; // ✅ Only match regular invoices (INV-XXXX), NOT refunds (REF-INV-XXXX) const match = invoice.match(/^INV-(\d+)$/); if (match) { const num = parseInt(match[1]); if (num > highestNum) { highestNum = num; } } } console.log(`📊 HIGHEST invoice number found: ${highestNum}`); } else { console.log('📊 No invoices found, starting at 1'); } // ✅ If no invoices found, start from 1 if (highestNum === 0) { highestNum = 0; } const nextNum = highestNum + 1; const newInvoice = `INV-${nextNum}`; console.log(`📋 NEW invoice number: ${newInvoice}`); return newInvoice; } catch (err) { console.error('❌ getNextIRSInvoiceNumber error:', err); const timestamp = Date.now(); return `INV-${timestamp}`; } }
